cmake_minimum_required(VERSION 3.20)
project(FlecsUnity LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(UNITY_PLUGIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Assets/Plugins/Flecs/x86_64" CACHE PATH "Unity native plugin output")
# -DFU_SANITIZE=ON builds the core, vendored libraries, plugin and tests with AddressSanitizer (plus UBSan where the
# compiler has it). It must be set before any target is created so every object file is instrumented.
option(FU_SANITIZE "Build with AddressSanitizer (MSVC /fsanitize=address; Clang/GCC address,undefined)" OFF)
if(FU_SANITIZE)
  if(MSVC)
    # ASan is incompatible with /RTC*, incremental linking and edit-and-continue (/ZI); /MT(d) is supported as is.
    foreach(flags CMAKE_C_FLAGS_DEBUG CMAKE_CXX_FLAGS_DEBUG)
      string(REGEX REPLACE "/RTC[1csu]+" "" ${flags} "${${flags}}")
      string(REPLACE "/ZI" "/Zi" ${flags} "${${flags}}")
    endforeach()
    add_compile_options(/fsanitize=address)
    foreach(kind EXE SHARED)
      string(APPEND CMAKE_${kind}_LINKER_FLAGS_DEBUG " /INCREMENTAL:NO")
    endforeach()
    add_link_options(/INCREMENTAL:NO)
  elseif(MINGW)
    message(FATAL_ERROR "FU_SANITIZE: this MinGW toolchain has no sanitizer runtime. Build with MSVC (Native/Generate-Solution.ps1) or Clang.")
  elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "FU_SANITIZE: unsupported compiler ${CMAKE_CXX_COMPILER_ID}.")
  endif()
endif()
add_subdirectory(Core)
add_subdirectory(App)
